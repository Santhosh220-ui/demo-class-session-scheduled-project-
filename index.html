Click "Test Connection" to verify if the Google Apps Script endpoint is open to public submissions or blocked by Google's permissions.

Why submissions don't show up in Google Sheets (401 Error Fix):

By default, Google Apps Script sets web app permissions to "Only myself". This causes Google to reject unauthenticated website submissions with HTTP 401 Unauthorized before the script can write to your spreadsheet.

  1. Open your spreadsheet (Spreadsheet ID: 1jBZ1tb...) and go to Extensions > Apps Script.
  2. Replace any code with the clean production script below, and click Save (Ctrl+S).
  3. Click the blue Deploy button at top right > Manage Deployments (or New Deployment).
  4. Under "Who has access", change it from "Only myself" to "Anyone" (This is mandatory so students can submit without logging in).
  5. Click Deploy. Copy the new Web App URL (ending in /exec), paste it into the URL field above, and click "Save & Use URL"!

/**
 * Pumo Technovation - Student Demo Registration Web App Backend
 * Google Spreadsheet ID: 1jBZ1tbLNVroOzcTTXbcRRBVk1G1hrQXJklHlig3nmbk
 * Version: 2.0 (Full Live Two-Way Sync: Read, Write & Auto-Delete)
 */

var SPREADSHEET_ID = "1jBZ1tbLNVroOzcTTXbcRRBVk1G1hrQXJklHlig3nmbk";

function getTargetSheet() {
  try {
    return SpreadsheetApp.openById(SPREADSHEET_ID).getActiveSheet();
  } catch (err) {
    return SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  }
}

/**
 * Handles GET requests:
 * 1. Default: Reads all live student demo rows directly from Google Sheet.
 * 2. action=delete: Deletes row matching referenceId or phone.
 * 3. action=clear: Clears all student rows (preserves Row 1 headers).
 */
function doGet(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var p = e ? (e.parameter || {}) : {};
    var action = p.action || "get";
    var sheet = getTargetSheet();
    var lastRow = sheet.getLastRow();

    // Action 1: Delete a specific record by Reference ID
    if (action === "delete") {
      var targetRefId = p.referenceId || p.refId || p.id || "";
      var deletedCount = 0;
      if (targetRefId && lastRow >= 2) {
        var colCount = Math.max(sheet.getLastColumn(), 7);
        var dataValues = sheet.getRange(2, 1, lastRow - 1, colCount).getValues();
        for (var r = dataValues.length - 1; r >= 0; r--) {
          var rowRef = String(dataValues[r][1] || "").trim();
          var rowPhone = String(dataValues[r][3] || "").replace(/\D/g, "");
          if (rowRef === targetRefId || (targetRefId.length === 10 && rowPhone === targetRefId)) {
            sheet.deleteRow(r + 2);
            deletedCount++;
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({
        result: "success",
        action: "delete",
        deletedCount: deletedCount,
        message: deletedCount > 0 ? ("Deleted " + deletedCount + " row(s) for " + targetRefId) : "No matching row found"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // Action 2: Clear all data rows (preserves headers)
    if (action === "clear" || action === "reset") {
      if (lastRow >= 2) {
        sheet.deleteRows(2, lastRow - 1);
      }
      return ContentService.createTextOutput(JSON.stringify({
        result: "success",
        action: "clear",
        message: "All student demo rows cleared from Google Sheet."
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // Action 3: GET All Current Records (Default)
    // If the sheet is empty or user manually deleted all rows in Google Sheets
    if (lastRow <= 1) {
      return ContentService.createTextOutput(JSON.stringify({
        result: "success",
        total: 0,
        records: [],
        message: "Sheet contains no data rows."
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var lastCol = Math.max(sheet.getLastColumn(), 7);
    var rangeValues = sheet.getRange(1, 1, lastRow, lastCol).getValues();
    var headers = rangeValues[0] || [];
    var records = [];

    // Identify column indices dynamically by header text
    var idxRef = 1;
    var idxName = 2;
    var idxContact = 3;
    var idxCourse = 4;
    var idxDate = 5;
    var idxStatus = 6;
    var idxTime = -1;
    var idxMode = -1;
    var idxTimestamp = 0;

    for (var h = 0; h < headers.length; h++) {
      var headerText = String(headers[h] || "").toLowerCase().trim();
      if (headerText.indexOf("ref") !== -1 || headerText.indexOf("id") !== -1) idxRef = h;
      else if (headerText.indexOf("name") !== -1 || headerText.indexOf("student") !== -1) idxName = h;
      else if (headerText.indexOf("contact") !== -1 || headerText.indexOf("mobile") !== -1 || headerText.indexOf("phone") !== -1) idxContact = h;
      else if (headerText.indexOf("course") !== -1 || headerText.indexOf("program") !== -1) idxCourse = h;
      else if (headerText.indexOf("date") !== -1) idxDate = h;
      else if (headerText.indexOf("time") !== -1 || headerText.indexOf("slot") !== -1) idxTime = h;
      else if (headerText.indexOf("mode") !== -1) idxMode = h;
      else if (headerText.indexOf("status") !== -1 || headerText.indexOf("convert") !== -1 || headerText.indexOf("remark") !== -1) idxStatus = h;
      else if (headerText.indexOf("timestamp") !== -1) idxTimestamp = h;
    }

    for (var i = 1; i < rangeValues.length; i++) {
      var row = rangeValues[i];
      var hasData = false;
      for (var c = 0; c < row.length; c++) {
        if (row[c] !== "" && row[c] !== null && row[c] !== undefined) {
          hasData = true;
          break;
        }
      }
      if (!hasData) continue;

      var rawName = row[idxName] ? String(row[idxName]).trim() : "";
      var rawContact = row[idxContact] ? String(row[idxContact]).replace(/^'/, "").trim() : "";
      var rawRef = row[idxRef] ? String(row[idxRef]).trim() : ("PT-DEMO-" + (i + 1000));
      var rawCourse = row[idxCourse] ? String(row[idxCourse]).trim() : "Python Full Stack";
      var rawDate = row[idxDate] ? String(row[idxDate]).trim() : "";
      var rawTime = (idxTime >= 0 && row[idxTime]) ? String(row[idxTime]).trim() : "Flexible Slot";
      var rawMode = (idxMode >= 0 && row[idxMode]) ? String(row[idxMode]).trim() : "Offline";
      var rawStatus = (idxStatus >= 0 && row[idxStatus]) ? String(row[idxStatus]).trim() : "Joining";
      var rawTimestamp = row[idxTimestamp] ? String(row[idxTimestamp]).trim() : new Date().toISOString();

      if (row[idxDate] instanceof Date) {
        var dt = row[idxDate];
        var yyyy = dt.getFullYear();
        var mm = String(dt.getMonth() + 1).padStart(2, "0");
        var dd = String(dt.getDate()).padStart(2, "0");
        rawDate = yyyy + "-" + mm + "-" + dd;
      }

      var cleanStatus = "Joining";
      if (rawStatus.indexOf("Not Converted") !== -1) {
        cleanStatus = "Not Converted";
      } else if (rawStatus.indexOf("Deciding") !== -1) {
        cleanStatus = "Deciding / Follow Up";
      }

      records.push({
        referenceId: rawRef,
        fullName: rawName,
        contactNumber: rawContact,
        courseName: rawCourse,
        courseTiming: rawTime,
        trainingMode: rawMode,
        demoDate: rawDate,
        joiningStatus: cleanStatus,
        readableTimestamp: rawTimestamp,
        submissionTimestamp: rawTimestamp
      });
    }

    return ContentService.createTextOutput(JSON.stringify({
      result: "success",
      total: records.length,
      records: records,
      spreadsheetId: SPREADSHEET_ID,
      lastUpdated: new Date().toISOString()
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      result: "error",
      message: err.toString(),
      records: []
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

/**
 * Handles POST requests: Appends new student registration row
 */
function doPost(e) {
  var lock = LockService.getScriptLock();
  var acquired = lock.tryLock(15000);

  try {
    var sheet = getTargetSheet();

    // Auto-create standard headers if row 1 is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp",
        "Reference ID",
        "Student Name",
        "Contact Number",
        "Course Name",
        "Course Timing",
        "Training Mode",
        "Date of Demo"
      ]);
      sheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#0d47a1").setFontColor("#ffffff");
      sheet.setFrozenRows(1);
    }

    var p = e.parameter || {};
    if ((!p.fullName && !p.studentName) && e.postData && e.postData.contents) {
      try {
        var parsed = JSON.parse(e.postData.contents);
        for (var k in parsed) p[k] = parsed[k];
      } catch (jsonErr) {}
    }

    var fullName = p.fullName || p.studentName || p.name || "N/A";
    var contact = p.contactNumber || p.contact || p.mobileNumber || p.phone || p.mobile || "N/A";
    var course = p.courseName || p.interestedCourse || p.course || "N/A";
    var timing = p.courseTiming || p.demoTime || p.time || p.slot || "10:00 AM - 11:30 AM (Morning Slot)";
    var mode = p.trainingMode || p.mode || "Offline";
    var demoDate = p.demoDate || p.date || "N/A";
    var refId = p.referenceId || ("PT-" + Math.floor(1000 + Math.random() * 9000));
    var timestamp = p.submissionDate || new Date().toLocaleString("en-IN", { timeZone: "Asia/Kolkata" });

    // If existing sheet has 16 columns from previous template, support both formats seamlessly
    if (sheet.getLastColumn() > 8) {
      sheet.appendRow([
        timestamp,
        refId,
        fullName,
        "'" + contact,
        "'" + contact,
        p.email || "N/A",
        p.collegeName || "N/A",
        p.degree || "N/A",
        p.department || "N/A",
        p.yearOfPassing || "N/A",
        course,
        demoDate,
        timing,
        mode,
        p.branchLocation || "Poonamallee",
        "Demo Registered"
      ]);
    } else {
      sheet.appendRow([
        timestamp,
        refId,
        fullName,
        "'" + contact,
        course,
        timing,
        mode,
        demoDate
      ]);
    }

    return ContentService.createTextOutput(JSON.stringify({
      result: "success",
      message: "Student registration saved to Google Sheet.",
      referenceId: refId,
      rowNumber: sheet.getLastRow()
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      result: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);

  } finally {
    if (acquired) lock.releaseLock();
  }
}
