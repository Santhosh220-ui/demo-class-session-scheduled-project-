Click "Test Connection" to verify if the Google Apps Script endpoint is open to public submissions or blocked by Google's permissions.

Why submissions don't show up in Google Sheets (401 Error Fix):

By default, Google Apps Script sets web app permissions to "Only myself". This causes Google to reject unauthenticated website submissions with HTTP 401 Unauthorized before the script can write to your spreadsheet.

  1. Open your spreadsheet (Spreadsheet ID: 1jBZ1tb...) and go to Extensions > Apps Script.
  2. Replace any code with the clean production script below, and click Save (Ctrl+S).
  3. Click the blue Deploy button at top right > Manage Deployments (or New Deployment).
  4. Under "Who has access", change it from "Only myself" to "Anyone" (This is mandatory so students can submit without logging in).
  5. Click Deploy. Copy the new Web App URL (ending in /exec), paste it into the URL field above, and click "Save & Use URL"!

/**
 * Pumo Technovation - Student Demo Registration Web App Backend
 * Google Spreadsheet ID: 1jBZ1tbLNVroOzcTTXbcRRBVk1G1hrQXJklHlig3nmbk
 */

var SPREADSHEET_ID = "1jBZ1tbLNVroOzcTTXbcRRBVk1G1hrQXJklHlig3nmbk";

function getTargetSheet() {
  try {
    return SpreadsheetApp.openById(SPREADSHEET_ID).getActiveSheet();
  } catch (err) {
    return SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  }
}

function doGet(e) {
  var responseData = {
    status: "online",
    message: "Pumo Technovation Google Sheets Registration API is active.",
    spreadsheetId: SPREADSHEET_ID,
    timestamp: new Date().toISOString()
  };
  return ContentService.createTextOutput(JSON.stringify(responseData))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  var acquired = lock.tryLock(15000);

  try {
    var sheet = getTargetSheet();

    // Auto-create standard headers if row 1 is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp",
        "Reference ID",
        "Student Full Name",
        "Mobile Number",
        "WhatsApp Number",
        "Email Address",
        "College Name",
        "Degree",
        "Department",
        "Year of Passing",
        "Interested Course",
        "Demo Date",
        "Demo Time",
        "Training Mode",
        "Branch Location",
        "Remarks"
      ]);
      sheet.getRange(1, 1, 1, 16).setFontWeight("bold").setBackground("#0d47a1").setFontColor("#ffffff");
      sheet.setFrozenRows(1);
    }

    var p = e.parameter || {};
    if ((!p.fullName && !p.studentName) && e.postData && e.postData.contents) {
      try {
        var parsed = JSON.parse(e.postData.contents);
        for (var k in parsed) p[k] = parsed[k];
      } catch (jsonErr) {}
    }

    var fullName = p.fullName || p.studentName || p.name || "N/A";
    var mobile = p.mobileNumber || p.phone || p.mobile || "N/A";
    var whatsapp = p.whatsappNumber || p.whatsapp || mobile;
    var email = p.email || p.emailAddress || "N/A";
    var college = p.collegeName || p.college || "N/A";
    var degree = p.degree || "N/A";
    var department = p.department || p.dept || "N/A";
    var yop = p.yearOfPassing || p.yop || "N/A";
    var course = p.interestedCourse || p.course || "N/A";
    var demoDate = p.demoDate || p.date || "N/A";
    var demoTime = p.demoTime || p.time || p.slot || "N/A";
    var mode = p.trainingMode || p.mode || "Offline";
    var branch = p.branchLocation || p.branch || p.location || "Poonamallee";
    var remarks = p.remarks || p.comments || "";
    var refId = p.referenceId || ("PT-" + Math.floor(1000 + Math.random() * 9000));
    var timestamp = p.submissionDate || new Date().toLocaleString("en-IN", { timeZone: "Asia/Kolkata" });

    sheet.appendRow([
      timestamp,
      refId,
      fullName,
      "'" + mobile,
      "'" + whatsapp,
      email,
      college,
      degree,
      department,
      yop,
      course,
      demoDate,
      demoTime,
      mode,
      branch,
      remarks
    ]);

    return ContentService.createTextOutput(JSON.stringify({
      result: "success",
      message: "Student registration saved to Google Sheet.",
      referenceId: refId,
      rowNumber: sheet.getLastRow()
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      result: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);

  } finally {
    if (acquired) lock.releaseLock();
  }
}
